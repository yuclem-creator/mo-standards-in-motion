First-time setup SQL 
create table sim_courses (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  status text not null default 'draft',
  data jsonb not null,
  updated_at timestamptz not null default now(),
  created_by uuid default auth.uid()
);

alter table sim_courses enable row level security;

create policy "authors full access"
  on sim_courses for all to authenticated
  using (true) with check (true);

create policy "anon read published"
  on sim_courses for select to anon
  using (status = 'published');

-- Storage: create a PUBLIC bucket named  sim-media
-- with insert restricted to authenticated authors.

-- Per-user, per-question results pool (one row per answer).
-- The SCORM player streams rows here; export to CSV for AI analysis.
create table sim_assessment_rows (
  id bigint generated always as identity primary key,
  at timestamptz not null default now(),
  course text,
  entry text,
  user_id text,
  user_name text,
  kind text,            -- question | exercise | summary
  interaction_id text,  -- e.g. assess-q2-try3, pre-q1, assessment-try-2
  chosen text,
  correct_answer text,
  result text,          -- correct | wrong | neutral
  score int,
  try int,
  passed boolean
);

alter table sim_assessment_rows enable row level security;

-- colleagues insert their own rows; only authors can read
create policy "anon insert results"
  on sim_assessment_rows for insert to anon
  with check (true);

create policy "authors read results"
  on sim_assessment_rows for select to authenticated
  using (true);
